cmake_minimum_required(VERSION 3.20)
project(ConvolverConsumer LANGUAGES CXX)
if(NOT CONVOLVER_CONSUMER_VERSION)
    message(FATAL_ERROR "Set CONVOLVER_CONSUMER_VERSION to the package major.minor under test")
endif()
option(CONSUMER_CTEST "Parent uses CTest" ON)
if(CONSUMER_CTEST)
    include(CTest)
else()
    enable_testing()
endif()
option(CONSUMER_IO "Exercise the optional I/O component" OFF)
if(CONVOLVER_SOURCE_DIR)
    set(CONVOLVER_BUILD_IO ${CONSUMER_IO} CACHE BOOL "" FORCE)
    set(CONVOLVER_BUILD_CLI OFF CACHE BOOL "" FORCE)
    add_subdirectory("${CONVOLVER_SOURCE_DIR}" convolver)
    if(TARGET convolver_engine_tests OR TARGET check)
        message(FATAL_ERROR "Subproject unexpectedly enabled Convolver tests")
    endif()
    if(NOT CONSUMER_CTEST AND DEFINED BUILD_TESTING)
        message(FATAL_ERROR "Subproject changed the parent's BUILD_TESTING policy")
    endif()
elseif(MISSING_COMPONENT)
    find_package(Convolver ${CONVOLVER_CONSUMER_VERSION} QUIET COMPONENTS ${MISSING_COMPONENT})
    if(Convolver_FOUND)
        message(FATAL_ERROR "Accepted missing required component ${MISSING_COMPONENT}")
    endif()
    # REL-19: a consumer that asked for the host layer installed up to 0.19
    # is told why it is gone and what to link instead.
    if(MISSING_COMPONENT STREQUAL "host" AND
       NOT Convolver_NOT_FOUND_MESSAGE MATCHES "no longer installs the 'host' component.*Convolver::core")
        message(FATAL_ERROR "No explanation for the removed host component: ${Convolver_NOT_FOUND_MESSAGE}")
    endif()
    if(MISSING_MESSAGE AND NOT Convolver_NOT_FOUND_MESSAGE MATCHES "${MISSING_MESSAGE}")
        message(FATAL_ERROR "Expected a message matching '${MISSING_MESSAGE}', got: ${Convolver_NOT_FOUND_MESSAGE}")
    endif()
    return()
else()
    find_package(Convolver ${CONVOLVER_CONSUMER_VERSION} REQUIRED COMPONENTS core)
    # Optional unknown components must not invalidate the usable core package.
    find_package(Convolver ${CONVOLVER_CONSUMER_VERSION} REQUIRED OPTIONAL_COMPONENTS nonexistent)
    if(CONSUMER_IO)
        find_package(Convolver ${CONVOLVER_CONSUMER_VERSION} REQUIRED COMPONENTS io)
        # IO-20: the .cfg loader is inside Convolver::io; the package
        # exports no other target.
        if(TARGET Convolver::configuration)
            message(FATAL_ERROR "The package exports Convolver::configuration, an internal library")
        endif()
    elseif(TARGET SndFile::sndfile OR TARGET convolver_io)
        message(FATAL_ERROR "Core-only consumption pulled in libsndfile/I/O")
    endif()
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE Convolver::core)
if(CONSUMER_IO)
    target_link_libraries(consumer PRIVATE Convolver::io)
    target_compile_definitions(consumer PRIVATE CONSUMER_IO)
endif()
add_test(NAME consumer.convolution COMMAND consumer)
# REL-19 (jrp's acceptance test for the 1.0 library): an installed consumer
# builds from the installed headers alone. None of the include directories
# the package gives is Convolver's source include directory (the build, and
# so this test's installation, may be inside the source tree), and the
# plug-ins' host header is not among the installed ones (main.cpp checks it).
if(NOT CONVOLVER_SOURCE_DIR)
    target_compile_definitions(consumer PRIVATE CONSUMER_INSTALLED)
    if(CONVOLVER_FORBIDDEN_SOURCE)
        file(REAL_PATH "${CONVOLVER_FORBIDDEN_SOURCE}" forbidden)
        foreach(target IN ITEMS Convolver::core Convolver::io)
            if(TARGET ${target})
                get_target_property(directories ${target} INTERFACE_INCLUDE_DIRECTORIES)
                foreach(directory IN LISTS directories)
                    file(REAL_PATH "${directory}" real)
                    string(FIND "${real}/" "${forbidden}/" at)
                    if(at EQUAL 0)
                        message(FATAL_ERROR "${target} includes ${directory}, Convolver's source headers")
                    endif()
                endforeach()
            endif()
        endforeach()
    endif()
endif()
