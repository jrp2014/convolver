cmake_minimum_required(VERSION 3.20)
project(ConvolverConsumer LANGUAGES CXX)
if(NOT CONVOLVER_CONSUMER_VERSION)
    message(FATAL_ERROR "Set CONVOLVER_CONSUMER_VERSION to the package major.minor under test")
endif()
option(CONSUMER_CTEST "Parent uses CTest" ON)
if(CONSUMER_CTEST)
    include(CTest)
else()
    enable_testing()
endif()
option(CONSUMER_HOST "Exercise portable host layer" OFF)
option(CONSUMER_IO "Exercise the optional I/O component" OFF)
if(CONVOLVER_SOURCE_DIR)
    set(CONVOLVER_BUILD_IO ${CONSUMER_IO} CACHE BOOL "" FORCE)
    set(CONVOLVER_BUILD_CLI OFF CACHE BOOL "" FORCE)
    add_subdirectory("${CONVOLVER_SOURCE_DIR}" convolver)
    if(TARGET convolver_engine_tests OR TARGET check)
        message(FATAL_ERROR "Subproject unexpectedly enabled Convolver tests")
    endif()
    if(NOT CONSUMER_CTEST AND DEFINED BUILD_TESTING)
        message(FATAL_ERROR "Subproject changed the parent's BUILD_TESTING policy")
    endif()
elseif(MISSING_COMPONENT)
    find_package(Convolver ${CONVOLVER_CONSUMER_VERSION} QUIET COMPONENTS ${MISSING_COMPONENT})
    if(Convolver_FOUND)
        message(FATAL_ERROR "Accepted missing required component ${MISSING_COMPONENT}")
    endif()
    return()
else()
    find_package(Convolver ${CONVOLVER_CONSUMER_VERSION} REQUIRED COMPONENTS core)
    # Optional unknown components must not invalidate the usable core package.
    find_package(Convolver ${CONVOLVER_CONSUMER_VERSION} REQUIRED OPTIONAL_COMPONENTS nonexistent)
    if(CONSUMER_HOST)
        find_package(Convolver ${CONVOLVER_CONSUMER_VERSION} REQUIRED COMPONENTS host)
    endif()
    if(CONSUMER_IO)
        find_package(Convolver ${CONVOLVER_CONSUMER_VERSION} REQUIRED COMPONENTS io)
    elseif(TARGET SndFile::sndfile OR TARGET convolver_io)
        message(FATAL_ERROR "Core-only consumption pulled in libsndfile/I/O")
    endif()
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE Convolver::core)
if(CONSUMER_IO)
    target_link_libraries(consumer PRIVATE Convolver::io)
    target_compile_definitions(consumer PRIVATE CONSUMER_IO)
endif()
add_test(NAME consumer.convolution COMMAND consumer)

if(CONSUMER_HOST)
    target_link_libraries(consumer PRIVATE Convolver::host)
    target_compile_definitions(consumer PRIVATE CONSUMER_HOST)
endif()
